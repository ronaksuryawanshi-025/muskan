        <h2>Website Visitors</h2>

        <canvas id="visitorChart"></canvas>

        <button onclick="addVisitorData()">
            Add New Month
        </button>

    <!-- JavaScript -->

    <script>

        let months = [
            "January",
            "February",
            "March",
            "April",
            "May",
            "June"
        ];

        let visitors = [
            120,
            180,
            250,
            210,
            320,
            400
        ];


        // Create the chart

        const ctx = document.getElementById("visitorChart");

        const visitorChart = new Chart(ctx, {

            type: "line",

            data: {
                labels: months,

                datasets: [{
                    label: "Website Visitors",
                    data: visitors,
                    borderWidth: 3,
                    tension: 0.3,
                    fill: false
                }]
            },

            options: {

                responsive: true,

                scales: {

                    y: {
                        beginAtZero: true,

                        title: {
                            display: true,
                            text: "Number of Visitors"
                        }
                    },

                    x: {
                        title: {
                            display: true,
                            text: "Month"
                        }
                    }

                }

            }

        });


        // Add new data dynamically

        function addVisitorData() {

            let newMonth = "July";
            let newVisitors = 520;

            months.push(newMonth);
            visitors.push(newVisitors);

            visitorChart.update();
        }

    </script>
